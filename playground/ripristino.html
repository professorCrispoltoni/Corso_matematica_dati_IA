# 1. dati finti: 3 feature, una categorica; target costruito da noi
import numpy as np, pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_absolute_error

rng = np.random.default_rng(0)
n = 500
tabella = pd.DataFrame({
    "distanza_km": rng.uniform(1, 40, n),
    "tipo_rete":   rng.choice(["aerea", "interrata"], n),
    "squadre":     rng.integers(1, 6, n)})
# regole inventate: base 30 min, +2 min per km, +25 se interrata, -5 per squadra
tabella["durata_min"] = (30 + 2.0 * tabella.distanza_km
                         + np.where(tabella.tipo_rete == "interrata", 25, 0)
                         - 5 * tabella.squadre + rng.normal(0, 8, n))

# 2. categoriche -> numeri (one-hot, una colonna in meno)
X = pd.get_dummies(tabella.drop(columns="durata_min"), drop_first=True, dtype=int)
y = tabella["durata_min"]

# 3. training e test; media e deviazione standard dal solo training
X_tr, X_te, y_tr, y_te = train_test_split(X, y, test_size=0.2, random_state=1)
sc = StandardScaler().fit(X_tr)

# 4. modello e valutazione
modello = LinearRegression().fit(sc.transform(X_tr), y_tr)
print("Errore medio sul test: %.1f minuti" % mean_absolute_error(y_te, modello.predict(sc.transform(X_te))))
