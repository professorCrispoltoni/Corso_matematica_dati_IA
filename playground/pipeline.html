Lo stesso procedimento in Python

# 1-2. X: le feature (n × p), y: 0 = sano, 1 = guasto
from sklearn.model_selection import train_test_split
from sklearn.pipeline import make_pipeline
from sklearn.preprocessing import StandardScaler
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (confusion_matrix,
                             classification_report)

X_tr, X_te, y_tr, y_te = train_test_split(
    X, y, test_size=0.2, stratify=y, random_state=42)

modello = make_pipeline(StandardScaler(), LogisticRegression())
modello.fit(X_tr, y_tr)        # μ, σ e pesi: solo dal training

y_pred = modello.predict(X_te)
print(confusion_matrix(y_te, y_pred))
print(classification_report(y_te, y_pred,
      target_names=["sano", "guasto"]))
